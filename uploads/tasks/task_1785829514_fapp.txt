  Future<List<dynamic>> _fetchEmployeeAttendance(
    int employeeId,
    String startDate,
    String endDate,
  ) async {
    try {
      final res = await http.get(
        Uri.parse(
          ApiConfig.fullFilterAttendanceUrl(employeeId, startDate, endDate),
        ),
        headers: {
          'Authorization': 'Bearer ${ApiConfig.token}',
          'Content-Type': 'application/json',
        },
      );
      if (res.statusCode == 200) {
        final data = jsonDecode(res.body);
        return data['attendances'] ?? [];
      }
    } catch (e) {
      debugPrint('Error filtering attendance: $e');
    }
    return [];
  }

  // ✅ Helper: Calculate working minutes between two DateTime strings
  int _getWorkingMinutes(String? checkInStr, String? checkOutStr) {
    if (checkInStr == null || checkOutStr == null || checkInStr.isEmpty || checkOutStr.isEmpty) return 0;
    try {
      final checkIn = DateTime.parse(checkInStr);
      final checkOut = DateTime.parse(checkOutStr);
      return checkOut.difference(checkIn).inMinutes;
    } catch (_) {
      return 0;
    }
  }

  // ✅ Helper: Calculate expected daily minutes from time strings
  int _getExpectedDailyMinutes(String? expectedArrival, String? expectedLeaving) {
    if (expectedArrival == null || expectedLeaving == null) return 0;
    try {
      final arrParts = expectedArrival.split(':');
      final levParts = expectedLeaving.split(':');
      if (arrParts.length < 2 || levParts.length < 2) return 0;
      final arrH = int.parse(arrParts[0]);
      final arrM = int.parse(arrParts[1]);
      final levH = int.parse(levParts[0]);
      final levM = int.parse(levParts[1]);
      return (levH * 60 + levM) - (arrH * 60 + arrM);
    } catch (_) {
      return 0;
    }
  }

  // ✅ Helper: Format minutes to readable string
  String _formatMinutes(int totalMinutes) {
    if (totalMinutes <= 0) return '0m';
    final hours = totalMinutes ~/ 60;
    final mins = totalMinutes % 60;
    if (hours == 0) return '${mins}m';
    if (mins == 0) return '${hours}h';
    return '${hours}h ${mins}m';
  }

                        final attendances = snapshot.data!;
                        final startDate = DateTime.parse(startDateStr);

                        final today = DateTime.now();
                        final todayOnly = DateTime(today.year, today.month, today.day);

                        // ✅ Check if a day is off
                        bool isOffDay(int weekday, String role) {
                          if (weekday == DateTime.sunday) return true;
                          if (weekday == DateTime.saturday && (role == 'intern' || role == 'student')) return true;

                        final currentDay = today.day;

                        List<Map<String, dynamic>> allWorkingDaysPassed = [];
                        for (int d = 1; d <= currentDay; d++) {
                          final dateObj = DateTime(now.year, now.month, d);
                          final dayOfWeek = dateObj.weekday;
                          if (!isOffDay(dayOfWeek, employeeRole)) {
                            workingDaysPassed++;
                            allWorkingDaysPassed.add({

                              'date': '${now.year}-${now.month.toString().padLeft(2, '0')}-${d.toString().padLeft(2, '0')}',

                        }

                        for (int d = 1; d <= totalDaysInMonth; d++) {
                          final dayOfWeek = DateTime(now.year, now.month, d).weekday;
                          if (!isOffDay(dayOfWeek, employeeRole)) totalWorkingDaysInMonth++;

                        for (var att in attendances) {
                          final checkInRaw = att['check_in']?.toString() ?? '';

                            try {

                        final actualWorkingDays = workingDaysSet.length;

                        for (var day in allWorkingDaysPassed) {

                        int totalGrossMinutes = 0;
                        
                        for (var att in attendances) {
                          final checkInRaw = att['check_in']?.toString() ?? '';
                          final checkOutRaw = att['check_out']?.toString() ?? att['company_check_out']?.toString() ?? '';
                          final pauseMinutes = int.tryParse(att['total_pause_minutes']?.toString() ?? '0') ?? 0;
                          
                          if (checkInRaw.isNotEmpty && checkOutRaw.isNotEmpty) {
                            final grossMinutes = _getWorkingMinutes(checkInRaw, checkOutRaw);
                            totalGrossMinutes += grossMinutes;
                            totalPauseMinutesMonth += pauseMinutes;
                            totalWorkedMinutes += (grossMinutes - pauseMinutes).clamp(0, 999999);
                          }
                        }

                        final weekdaysShort = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];

                        String formatLeaveDate(Map<String, dynamic> day) {
                          return '${day['dayName']} ${day['day']}';

                                    Row(

                                        Text("${member.name}'s Monthly Overview", style: TextStyle(fontSize: 11.5.sp, fontWeight: FontWeight.bold, color: AppColors.textPrimary)),
                                      ],
                                    ),
                                    SizedBox(height: 8.h),
                                    // Row 1
                                    Row(
                                      children: [
                                        Expanded(child: _buildStatCard('WORKING DAYS IN MONTH', '$totalWorkingDaysInMonth days', AppColors.textPrimary)),
                                        SizedBox(width: 6.w),
                                        Expanded(child: _buildStatCard('TOTAL EXPECTED HOURS', _formatMinutes(totalExpectedMonthMinutes), AppColors.textPrimary)),
                                      ],
                                    ),
                                    SizedBox(height: 5.h),
                                    // Row 2
                                    Row(
                                      children: [
                                        Expanded(child: _buildStatCard('TOTAL LEAVES', '$totalLeaves ${totalLeaves == 1 ? 'day' : 'days'}', totalLeaves > 0 ? const Color(0xFFDC2626) : const Color(0xFF16A34A))),
                                        SizedBox(width: 6.w),
                                        Expanded(child: _buildStatCard('NET HOURS WORKED', _formatMinutes(totalWorkedMinutes), const Color(0xFF10B981))),
                                      ],
                                    ),
                                    SizedBox(height: 5.h),
                                    // Row 3
                                    Row(
                                      children: [
                                        Expanded(child: _buildStatCard('DAYS WORKED / TOTAL', '$actualDaysWorked / $totalWorkingDaysInMonth', const Color(0xFFD97706))),
                                        SizedBox(width: 6.w),
                                        Expanded(child: _buildStatCard('ACTUAL HOURS / EXPECTED', '${_formatMinutes(totalWorkedMinutes)} / ${_formatMinutes(currentExpectedMinutes)}', const Color(0xFFD97706))),

                                    SizedBox(height: 5.h),
                                    // Row 4: Break time or Daily Expected

                                      children: [
                                        Expanded(
                                          child: totalPauseMinutesMonth > 0
                                              ? _buildStatCard('TOTAL BREAK TIME', _formatMinutes(totalPauseMinutesMonth), const Color(0xFFF59E0B))

                                          child: totalPauseMinutesMonth > 0
                                              ? _buildStatCard('GROSS (INCL. BREAKS)', _formatMinutes(totalGrossMinutes), const Color(0xFF64748B))

                                    // Row 5: Net hours when pauses exist
                                    if (totalPauseMinutesMonth > 0) ...[
                                      SizedBox(height: 5.h),
                                      Row(
                                        children: [
                                          Expanded(child: _buildStatCard('NET HOURS (ACTUAL)', _formatMinutes(totalWorkedMinutes), const Color(0xFF10B981))),
